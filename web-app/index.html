전체 소스 코드는 함께 제공된 ZIP 패키지(`voca_arch_full_project.zip`)에서도 다운로드하실 수 있습니다.

1. Supabase PostgreSQL DDL (`vocab_entries`)
CREATE TABLE public.vocab_entries (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  book_title TEXT DEFAULT NULL,   -- 비워둘 수 있음
  chapter_name TEXT DEFAULT NULL, -- 비워둘 수 있음
  word TEXT NOT NULL,
  phonetic TEXT DEFAULT NULL,     -- 없는 경우 API/내장사전 보강
  audio_url TEXT DEFAULT NULL,    -- 커스텀 음성 또는 사전 MP3
  part_of_speech TEXT NOT NULL,   -- 다품사 행 분리 기준
  meaning_ko TEXT NOT NULL,
  example_sentence TEXT DEFAULT NULL,
  attempt_count INT NOT NULL DEFAULT 0,
  wrong_count INT NOT NULL DEFAULT 0,
  error_rate NUMERIC(5,2) GENERATED ALWAYS AS (
    CASE WHEN attempt_count > 0
      THEN ROUND((wrong_count::NUMERIC / attempt_count) * 100, 2)
      ELSE 0.00 END
  ) STORED,
  seq_order BIGINT GENERATED BY DEFAULT AS IDENTITY
);
